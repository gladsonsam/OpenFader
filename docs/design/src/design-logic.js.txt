<script type="text/x-dc" data-dc-script="" data-props="{&quot;stripsPerScreen&quot;:{&quot;editor&quot;:&quot;enum&quot;,&quot;options&quot;:[&quot;16&quot;,&quot;12&quot;,&quot;8&quot;],&quot;default&quot;:&quot;12&quot;},&quot;$preview&quot;:{&quot;width&quot;:1280,&quot;height&quot;:720}}">
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.levels = {};
    this.clips = {};
    this.peaks = {};
    this.drag = null;
    this._tc = {};
    this.taps = [];
    this.state = {
      mode: 'mixer', prevMode: 'mixer', tab: 'eq', bank: 'ch1', sel: 'in1', selBand: 1,
      fine: false, muteEnable: true, sof: false, mix: 0, tick: 0, scene: 4, scenePv: 4,
      spp: null, peakHold: true, locked: false, sceneConfirm: true, phantomArm: null,
      fxSel: 0, fx: this.buildFx(),
      patchSel: 'in1', patchGroup: 'Lcl', routeBank: 0,
      connected: 'foh', lastScan: 'Scanned at start-up',
      chans: this.buildChannels()
    };
  }

  componentDidMount() { this.timer = setInterval(() => this.stepMeters(), 80); }
  componentWillUnmount() { clearInterval(this.timer); clearTimeout(this.armTimer); }

  // ---------- data ----------
  bankDefs() {
    if (this._banks) return this._banks;
    const range = (n, f) => Array.from({ length: n }, (_, i) => f(i));
    const pad = n => String(n).padStart(2, '0');
    const G = '#5cdb5c', GT = '#10240f';
    this._banks = [
      { key: 'ch1', label: 'Ch 1-16', color: G, text: GT, items: range(16, i => ({ id: 'in' + (i + 1), name: 'Ch ' + (i + 1), src: 'Lcl ' + pad(i + 1), eq: true, kind: 'in' })) },
      { key: 'ch17', label: 'Ch 17-32', color: G, text: GT, items: range(16, i => ({ id: 'in' + (i + 17), name: 'Ch ' + (i + 17), src: 'Lcl ' + pad(i + 17), eq: true, kind: 'in' })) },
      { key: 'ch33', label: 'Ch 33-48', color: G, text: GT, items: range(16, i => ({ id: 'in' + (i + 33), name: 'Ch ' + (i + 33), src: 'Net ' + pad(i + 1), eq: true, kind: 'in' })) },
      { key: 'fx', label: 'FxRet', color: G, text: GT, items: range(16, i => ({ id: 'fx' + (i + 1), name: 'FxRet ' + (i + 1), src: 'FX ' + (Math.floor(i / 2) + 1) + (i % 2 ? 'R' : 'L'), eq: true, kind: 'ret' })) },
      { key: 'mix', label: 'Mix', color: '#22d3e6', text: '#062a30', items: range(16, i => ({ id: 'mix' + (i + 1), name: 'Mix ' + (i + 1), src: 'Bus', eq: true, kind: 'bus' })) },
      { key: 'grp', label: 'Grp/FxSnd', color: '#2563eb', text: '#ffffff', items: range(8, i => ({ id: 'grp' + (i + 1), name: 'Grp ' + (i + 1), src: 'Bus', eq: true, kind: 'bus' })).concat(range(8, i => ({ id: 'fxs' + (i + 1), name: 'FxSnd ' + (i + 1), src: 'FX ' + (i + 1), eq: true, kind: 'bus' }))) },
      { key: 'mtx', label: 'Mtx/Main', color: '#c026d3', text: '#ffffff', items: range(8, i => ({ id: 'mtx' + (i + 1), name: 'Mtx ' + (i + 1), src: 'Out', eq: true, kind: 'out' })).concat([{ id: 'mainlr', name: 'Main LR', src: 'Out', eq: true, kind: 'out' }, { id: 'mainc', name: 'Main C', src: 'Out', eq: true, kind: 'out' }]) },
      { key: 'dca', label: 'DCA', color: '#dc2626', text: '#ffffff', items: range(16, i => ({ id: 'dca' + (i + 1), name: 'DCA ' + (i + 1), src: 'DCA', eq: false, kind: 'dca' })) }
    ];
    return this._banks;
  }

  buildChannels() {
    let seed = 7;
    const r = () => { seed = (seed * 16807) % 2147483647; return (seed - 1) / 2147483646; };
    const half = v => Math.round(v * 2) / 2;
    const out = {};
    this.bankDefs().forEach(b => b.items.forEach(it => {
      const isIn = it.kind === 'in' || it.kind === 'ret';
      out[it.id] = {
        id: it.id, name: it.name, src: it.src, color: b.color, text: b.text, kind: it.kind,
        act: it.kind === 'dca' ? 0.55 : (r() < 0.15 ? 0.05 : 0.35 + r() * 0.62),
        ph: r() * 6,
        fader: it.kind === 'dca' ? 0.75 : 0.6 + r() * 0.15,
        mute: false,
        pan: (it.kind === 'dca' || it.kind === 'out') ? null : (r() < 0.5 ? 0 : Math.round((r() * 2 - 1) * 60) / 100),
        sends: isIn ? Array.from({ length: 16 }, () => Math.round(r() * 70) / 100) : null,
        sendPre: isIn ? Array.from({ length: 16 }, (_, k) => k < 8) : null,
        input: it.kind === 'in' ? { gain: Math.round(20 + r() * 30), trim: 0, phantom: false, invert: false, delayOn: false, delay: 0 } : null,
        eq: it.eq ? {
          on: true,
          hpf: { on: it.kind === 'in' && r() < 0.7, f: 60 + Math.round(r() * 80), slope: 12 },
          bands: [
            { type: 'LS', f: 120, g: r() < 0.4 ? half(r() * 8 - 4) : 0, q: 0.7, on: true },
            { type: 'PEQ', f: Math.round(250 + r() * 400), g: r() < 0.5 ? -half(r() * 6) : 0, q: 1.4, on: true },
            { type: 'PEQ', f: Math.round(1800 + r() * 2500), g: r() < 0.5 ? half(r() * 5) : 0, q: Math.round((1.2 + r()) * 100) / 100, on: true },
            { type: 'HS', f: 8000, g: r() < 0.4 ? half(r() * 4) : 0, q: 0.7, on: true }
          ]
        } : null,
        dyn: it.eq ? {
          gate: { on: r() < 0.3, thr: Math.round(-55 + r() * 15), range: 40, att: 0.5, hold: 50, rel: 150 },
          comp: { on: r() < 0.7, thr: Math.round(-28 + r() * 16), ratio: [2, 3, 4, 6][Math.floor(r() * 4)], att: 10, rel: 120, knee: 3, makeup: 0 }
        } : null
      };
    }));
    return out;
  }

  buildFx() {
    const T = ['Hall', 'Plate', 'Delay', 'Chorus', 'Room', 'Delay', 'Flanger', 'Hall'];
    const P = { Hall: [0.42, 0.15, 0.6, 0.55, 1], Plate: [0.25, 0.08, 0.5, 0.65, 1], Room: [0.1, 0.05, 0.3, 0.6, 1], Delay: [0.55, 0.35, 0.55, 0.4, 1], Chorus: [0.4, 0.5, 0.6, 1, 1], Flanger: [0.3, 0.6, 0.4, 1, 1] };
    return T.map((t, i) => ({ type: t, on: i !== 7, p: P[t].slice(), bpm: null }));
  }
  fxDefs(type) {
    const pct = v => Math.round(v * 100) + '%';
    const hz = v => this.fmtF(1000 * Math.pow(20, v));
    const rev = [['Decay', v => (0.2 + v * 9.8).toFixed(1) + ' s'], ['Pre-delay', v => Math.round(v * 200) + ' ms'], ['Size', pct], ['Hi Damp', hz], ['Mix', pct]];
    const rate = ['Rate', v => (0.05 * Math.pow(100, v)).toFixed(2) + ' Hz'];
    return {
      Hall: rev, Plate: rev, Room: rev,
      Delay: [['Time', v => this.fmtMs(20 * Math.pow(100, v))], ['Feedback', v => Math.round(v * 95) + '%'], ['Hi Cut', hz], ['Spread', pct], ['Mix', pct]],
      Chorus: [rate, ['Depth', pct], ['Spread', pct], ['Mix', pct]],
      Flanger: [rate, ['Depth', pct], ['Feedback', v => Math.round(v * 90) + '%'], ['Mix', pct]]
    }[type];
  }
  scenes() { return ['Soundcheck', 'Doors', 'Walk-in', 'Support act', 'Changeover', 'Set 1', 'Acoustic', 'Set 2', 'Encore', 'Walk-out']; }
  devices() {
    return [
      { key: 'foh', name: 'FOH Console', model: '[Console model]', ip: '192.168.1.20', kind: 'console' },
      { key: 'mon', name: 'Monitor Console', model: '[Console model]', ip: '192.168.1.21', kind: 'console' },
      { key: 'sb1', name: 'Stage Box A', model: '[I/O model]', ip: '192.168.1.40', kind: 'io' }
    ];
  }

  // ---------- fader law ----------
  law() { return [[0, -90], [0.125, -50], [0.25, -30], [0.375, -20], [0.5, -10], [0.625, -5], [0.75, 0], [0.875, 5], [1, 10]]; }
  posToDb(p) {
    if (p <= 0.003) return -Infinity;
    const T = this.law();
    for (let i = 1; i < T.length; i++) if (p <= T[i][0]) { const a = T[i - 1], b = T[i]; return a[1] + (b[1] - a[1]) * (p - a[0]) / (b[0] - a[0]); }
    return 10;
  }
  dbToPos(db) {
    const T = this.law();
    if (db <= -90) return 0;
    for (let i = 1; i < T.length; i++) if (db <= T[i][1]) { const a = T[i - 1], b = T[i]; return a[0] + (b[0] - a[0]) * (db - a[1]) / (b[1] - a[1]); }
    return 1;
  }
  fmtDb(p) { const d = this.posToDb(p); if (!isFinite(d) || d < -89) return '-∞'; return (d > 0.04 ? '+' : '') + d.toFixed(1); }
  fmtF(f) { return f < 1000 ? Math.round(f) + ' Hz' : (f / 1000).toFixed(f < 10000 ? 2 : 1) + ' kHz'; }
  fmtG(g) { return (g > 0 ? '+' : '') + g.toFixed(1) + ' dB'; }
  fmtMs(v) { return v >= 1000 ? (v / 1000).toFixed(2) + ' s' : v < 10 ? v.toFixed(2) + ' ms' : v < 100 ? v.toFixed(1) + ' ms' : Math.round(v) + ' ms'; }
  clamp(v, a, b) { return Math.max(a, Math.min(b, v)); }

  // ---------- EQ math (RBJ biquads @ 48k) ----------
  coefs(type, f, g, q) {
    const fs = 48000, A = Math.pow(10, g / 40), w = 2 * Math.PI * f / fs, cs = Math.cos(w), sn = Math.sin(w), al = sn / (2 * q);
    let b0, b1, b2, a0, a1, a2;
    if (type === 'PEQ') { b0 = 1 + al * A; b1 = -2 * cs; b2 = 1 - al * A; a0 = 1 + al / A; a1 = -2 * cs; a2 = 1 - al / A; }
    else if (type === 'LS') { const s = 2 * Math.sqrt(A) * al; b0 = A * ((A + 1) - (A - 1) * cs + s); b1 = 2 * A * ((A - 1) - (A + 1) * cs); b2 = A * ((A + 1) - (A - 1) * cs - s); a0 = (A + 1) + (A - 1) * cs + s; a1 = -2 * ((A - 1) + (A + 1) * cs); a2 = (A + 1) + (A - 1) * cs - s; }
    else if (type === 'HS') { const s = 2 * Math.sqrt(A) * al; b0 = A * ((A + 1) + (A - 1) * cs + s); b1 = -2 * A * ((A - 1) + (A + 1) * cs); b2 = A * ((A + 1) + (A - 1) * cs - s); a0 = (A + 1) - (A - 1) * cs + s; a1 = 2 * ((A - 1) - (A + 1) * cs); a2 = (A + 1) - (A - 1) * cs - s; }
    else { b0 = (1 + cs) / 2; b1 = -(1 + cs); b2 = (1 + cs) / 2; a0 = 1 + al; a1 = -2 * cs; a2 = 1 - al; }
    return [b0 / a0, b1 / a0, b2 / a0, a1 / a0, a2 / a0];
  }
  magDb(c, f) {
    const w = 2 * Math.PI * f / 48000, c1 = Math.cos(w), s1 = Math.sin(w), c2 = Math.cos(2 * w), s2 = Math.sin(2 * w);
    const nr = c[0] + c[1] * c1 + c[2] * c2, ni = -(c[1] * s1 + c[2] * s2), dr = 1 + c[3] * c1 + c[4] * c2, di = -(c[3] * s1 + c[4] * s2);
    return 10 * Math.log10((nr * nr + ni * ni) / (dr * dr + di * di));
  }
  filters(eq) {
    const out = [];
    if (!eq || !eq.on) return out;
    if (eq.hpf && eq.hpf.on) { const c = this.coefs('HP', eq.hpf.f, 0, 0.7071); out.push(c); if (eq.hpf.slope === 24) out.push(c); }
    (eq.bands || []).forEach(b => { if (b.on) out.push(this.coefs(b.type, b.f, b.g, b.q)); });
    return out;
  }
  resp(fl, f) { let s = 0; for (let i = 0; i < fl.length; i++) s += this.magDb(fl[i], f); return s; }
  eqThumb(c) {
    if (!c.eq) return '';
    const k = JSON.stringify(c.eq);
    if (this._tc[c.id] && this._tc[c.id].k === k) return this._tc[c.id].p;
    const fl = this.filters(c.eq), pts = [];
    for (let i = 0; i <= 40; i++) { const f = 20 * Math.pow(1000, i / 40); const db = fl.length ? this.resp(fl, f) : 0; pts.push((i * 2.5).toFixed(1) + ' ' + Math.max(0, Math.min(40, 20 - db / 15 * 20)).toFixed(1)); }
    const p = 'M' + pts.join(' L');
    this._tc[c.id] = { k, p };
    return p;
  }

  // ---------- dynamics math ----------
  gateOut(g, x) { return g.on && x < g.thr ? x - g.range : x; }
  compOut(c, x) {
    if (!c.on) return x;
    const d = x - c.thr, k = c.knee;
    let y;
    if (2 * d < -k) y = x;
    else if (k > 0 && 2 * Math.abs(d) <= k) y = x + (1 / c.ratio - 1) * Math.pow(d + k / 2, 2) / (2 * k);
    else y = c.thr + d / c.ratio;
    return y + c.makeup;
  }
  dynThumb(c) {
    if (!c.dyn) return '';
    const pts = [];
    for (let i = 0; i <= 24; i++) {
      const inp = -60 + i * 2.5;
      const o = this.compOut(c.dyn.comp, this.gateOut(c.dyn.gate, inp));
      pts.push(((inp + 60) / 60 * 100).toFixed(1) + ' ' + Math.max(0, Math.min(40, -o / 60 * 40)).toFixed(1));
    }
    return 'M' + pts.join(' L');
  }
  spec(key) {
    const db = v => v.toFixed(1) + ' dB', ms = v => this.fmtMs(v);
    return {
      gain: { min: 0, max: 60, step: 0.5, fmt: db },
      trim: { min: -18, max: 18, step: 0.5, fmt: v => this.fmtG(v) },
      delay: { min: 0, max: 500, step: 0.1, fmt: ms },
      gthr: { min: -80, max: 0, step: 0.5, fmt: db },
      range: { min: 0, max: 80, step: 1, fmt: v => v + ' dB' },
      gatt: { min: 0.05, max: 50, log: true, fmt: ms },
      hold: { min: 1, max: 2000, log: true, fmt: ms },
      grel: { min: 5, max: 4000, log: true, fmt: ms },
      cthr: { min: -60, max: 0, step: 0.5, fmt: db },
      ratio: { min: 1, max: 20, log: true, fmt: v => v.toFixed(1) + ':1' },
      catt: { min: 0.1, max: 100, log: true, fmt: ms },
      crel: { min: 5, max: 2000, log: true, fmt: ms },
      knee: { min: 0, max: 12, step: 0.5, fmt: db },
      makeup: { min: 0, max: 24, step: 0.5, fmt: v => this.fmtG(v) }
    }[key];
  }
  toN(s, v) { return s.log ? Math.log(v / s.min) / Math.log(s.max / s.min) : (v - s.min) / (s.max - s.min); }
  fromN(s, n) {
    let v = s.log ? s.min * Math.pow(s.max / s.min, n) : s.min + n * (s.max - s.min);
    return s.step ? Math.round(v / s.step) * s.step : Math.round(v * 100) / 100;
  }

  // ---------- state helpers ----------
  updCh(id, fn) {
    this.setState(s => { const c = JSON.parse(JSON.stringify(s.chans[id])); fn(c); return { chans: Object.assign({}, s.chans, { [id]: c }) }; });
  }
  updFx(i, fn) {
    this.setState(s => { const fx = s.fx.map((x, k) => k === i ? JSON.parse(JSON.stringify(x)) : x); fn(fx[i]); return { fx }; });
  }
  openPage(m, extra) {
    const st = this.state;
    const pages = ['fx', 'routing', 'connect', 'settings', 'scenes'];
    const prev = pages.indexOf(st.mode) >= 0 ? st.prevMode : st.mode;
    this.setState(Object.assign({ mode: m, prevMode: prev }, extra || {}));
  }

  stepMeters() {
    const st = this.state, bank = this.bankDefs().find(b => b.key === st.bank);
    const ids = bank.items.map(i => i.id);
    [st.sel, 'mainlr', 'mix' + (st.mix + 1), 'fx1', 'fx3', 'fx5', 'fx7', 'fx9', 'fx11', 'fx13', 'fx15'].forEach(x => { if (ids.indexOf(x) < 0) ids.push(x); });
    const t = Date.now() / 1000, now = Date.now();
    ids.forEach(id => {
      const c = st.chans[id];
      let cur = this.levels[id] != null ? this.levels[id] : -90, target = -90;
      if (c.act > 0.12) {
        target = -42 + c.act * 40 + Math.sin(t * (1.3 + c.act * 2) + c.ph) * 6 + (Math.random() * 10 - 7);
        if (Math.random() < 0.06) target += 6;
      }
      cur = target > cur ? cur + (target - cur) * 0.7 : Math.max(-90, cur - 1.4);
      this.levels[id] = cur;
      if (cur > -0.3) this.clips[id] = now + 1500;
      const pk = this.peaks[id];
      if (!pk || cur >= pk.v || now > pk.until) this.peaks[id] = { v: cur, until: now + 1500 };
    });
    this.setState({ tick: (st.tick + 1) % 1000000 });
  }

  // ---------- pointer drags ----------
  capture(e) { try { e.currentTarget.setPointerCapture(e.pointerId); } catch (err) {} }
  faderDown(e, id) {
    this.capture(e);
    const c = this.state.chans[id], useSend = this.state.sof && !!c.sends;
    const h = Math.max(80, e.currentTarget.getBoundingClientRect().height - 60);
    this.drag = { kind: 'fader', id, useSend, y: e.clientY, start: useSend ? c.sends[this.state.mix] : c.fader, h };
  }
  panDown(e, id) {
    this.capture(e);
    this.drag = { kind: 'pan', id, x: e.clientX, start: this.state.chans[id].pan || 0, w: e.currentTarget.getBoundingClientRect().width };
  }
  knobDown(e, id, band, param) {
    this.capture(e);
    const eq = this.state.chans[id].eq;
    const v = band === 0 ? eq.hpf.f : eq.bands[band - 1][param];
    this.drag = { kind: 'knob', id, band, param, y: e.clientY, start: v };
    if (this.state.selBand !== band) this.setState({ selBand: band });
  }
  nodeDown(e, id, band) {
    this.capture(e);
    const g = e.currentTarget.closest('[data-graph]');
    this.drag = { kind: 'node', id, band, r: g.getBoundingClientRect() };
    if (this.state.selBand !== band) this.setState({ selBand: band });
  }
  move(e) {
    const d = this.drag;
    if (!d || this.state.locked) return;
    const fine = this.state.fine ? 0.25 : 1;
    if (d.kind === 'fader') {
      const v = this.clamp(d.start + (d.y - e.clientY) / d.h * fine, 0, 1);
      const mix = this.state.mix;
      this.updCh(d.id, c => { if (d.useSend) c.sends[mix] = v; else c.fader = v; });
    } else if (d.kind === 'pan') {
      let v = this.clamp(d.start + (e.clientX - d.x) / d.w * 2 * fine, -1, 1);
      if (Math.abs(v) < 0.03) v = 0;
      this.updCh(d.id, c => { c.pan = Math.round(v * 100) / 100; });
    } else if (d.kind === 'knob') {
      const dy = (d.y - e.clientY) * fine;
      let v;
      if (d.param === 'f') v = this.clamp(d.start * Math.pow(2, dy / 30), 20, d.band === 0 ? 1000 : 20000);
      else if (d.param === 'g') v = Math.round(this.clamp(d.start + dy * 0.1, -15, 15) * 10) / 10;
      else v = Math.round(this.clamp(d.start * Math.pow(2, dy / 60), 0.3, 16) * 100) / 100;
      this.updCh(d.id, c => { if (d.band === 0) c.eq.hpf.f = v; else c.eq.bands[d.band - 1][d.param] = v; });
    } else if (d.kind === 'node') {
      const x = this.clamp((e.clientX - d.r.left) / d.r.width, 0, 1), y = this.clamp((e.clientY - d.r.top) / d.r.height, 0, 1);
      const f = 20 * Math.pow(1000, x), g = Math.round((15 - 30 * y) * 10) / 10;
      this.updCh(d.id, c => { if (d.band === 0) c.eq.hpf.f = Math.min(1000, f); else { const b = c.eq.bands[d.band - 1]; b.f = f; b.g = g; } });
    } else if (d.kind === 'g') {
      d.set(this.clamp(d.start + (d.y - e.clientY) / 180 * fine, 0, 1));
    } else if (d.kind === 'h') {
      d.set(this.clamp(d.start + (e.clientX - d.x) / d.w * fine, 0, 1));
    }
  }
  end() { this.drag = null; }
  resetParam(id, band, param) {
    const F = [80, 100, 400, 2500, 8000], Q = [0.7, 0.7, 1.4, 1.4, 0.7];
    this.updCh(id, c => {
      if (band === 0) { c.eq.hpf.f = 80; return; }
      const b = c.eq.bands[band - 1];
      if (param === 'f') b.f = F[band]; else if (param === 'g') b.g = 0; else b.q = Q[band];
    });
  }
  tap(i) {
    const now = Date.now();
    this.taps = this.taps.filter(t => now - t < 2500);
    this.taps.push(now);
    if (this.taps.length >= 2) {
      const iv = (this.taps[this.taps.length - 1] - this.taps[0]) / (this.taps.length - 1);
      const bpm = Math.round(60000 / iv), ms = this.clamp(60000 / bpm, 20, 2000);
      this.updFx(i, s => { s.bpm = bpm; s.p[0] = Math.log(ms / 20) / Math.log(100); });
    }
  }

  // ---------- view helpers ----------
  knobItem(label, n, text, color, onDown, onReset) {
    n = this.clamp(n, 0, 1);
    return {
      label, value: text, aria: label + ' ' + text, color,
      dash: (n * 103.67).toFixed(2) + ' 138.23',
      rot: 'rotate(' + (-135 + n * 270).toFixed(1) + ' 28 28)',
      down: onDown, move: e => this.move(e), up: () => this.end(), reset: onReset
    };
  }
  chKnob(id, label, key, get, set, def, color) {
    const s = this.spec(key), v = get(this.state.chans[id]);
    return this.knobItem(label, this.toN(s, v), s.fmt(v), color,

  eqVals(c) {
    const st = this.state, W = 1000, H = 300, eq = c.eq, N = 180, id = c.id;
    const COL = ['#d7dde8', '#ff9f43', '#f6d743', '#3ed0e0', '#b28cff'];
    const NAMES = ['HPF', 'Low', 'Lo-Mid', 'Hi-Mid', 'High'];
    const toY = db => this.clamp((15 - db) / 30 * H, -20, H + 20);
    const xOf = f => Math.log(f / 20) / Math.log(1000);
    const pathFor = fl => { const pts = []; for (let i = 0; i <= N; i++) { const f = 20 * Math.pow(1000, i / N); pts.push((i / N * W).toFixed(1) + ' ' + toY(fl.length ? this.resp(fl, f) : 0).toFixed(1)); } return 'M' + pts.join(' L'); };
    const all = this.filters(eq);
    const curvePath = pathFor(all);
    const fillPath = curvePath + ' L' + W + ' ' + H / 2 + ' L0 ' + H / 2 + ' Z';
    const sb = st.selBand;
    let own = [];
    if (eq.on) {
      if (sb === 0 && eq.hpf.on) own = this.filters({ on: true, hpf: eq.hpf, bands: [] });
      if (sb > 0 && eq.bands[sb - 1].on) own = this.filters({ on: true, hpf: { on: false }, bands: [eq.bands[sb - 1]] });
    }
    const bandFill = own.length ? pathFor(own) + ' L' + W + ' ' + H / 2 + ' L0 ' + H / 2 + ' Z' : '';
    const gridV = [];
    [10, 100, 1000, 10000].forEach(dd => { for (let m = 1; m <= 9; m++) { const f = dd * m; if (f < 20 || f > 20000) continue; gridV.push({ x: (xOf(f) * W).toFixed(1), stroke: (m === 1 || m === 2 || m === 5) ? '#223047' : '#141c2a' }); } });
    const gridH = [12, 6, 0, -6, -12].map(g => ({ y: toY(g).toFixed(1), stroke: g === 0 ? '#34445e' : '#1b2536' }));
    const freqLabels = [[50, '50'], [100, '100'], [200, '200'], [500, '500'], [1000, '1k'], [2000, '2k'], [5000, '5k'], [10000, '10k']].map(a => ({ x: (xOf(a[0]) * 100).toFixed(2), label: a[1] }));
    const gainLabels = [12, 6, 0, -6, -12].map(g => ({ y: ((15 - g) / 30 * 100).toFixed(2), label: (g > 0 ? '+' : '') + g }));
    const nodes = [];
    const mkNode = (band, f, g, on) => {
      const active = sb === band;
      nodes.push({
        label: band === 0 ? 'H' : String(band), aria: NAMES[band] + ' node, ' + this.fmtF(f) + (band ? ', ' + this.fmtG(g) : ''),
        x: (this.clamp(xOf(f), 0, 1) * 100).toFixed(2), y: (this.clamp((15 - g) / 30, 0.03, 0.97) * 100).toFixed(2),
        size: active ? 36 : 30, ring: active ? '#ffffff' : 'rgba(0,0,0,.55)', color: COL[band], op: on && eq.on ? 1 : 0.4,
        down: e => this.nodeDown(e, id, band), move: e => this.move(e), up: () => this.end()
      });
    };
    mkNode(0, eq.hpf.f, -3, eq.hpf.on);
    eq.bands.forEach((b, i) => mkNode(i + 1, b.f, b.g, b.on));
    const chip = (label, active, pick) => ({ label, pressed: active ? 'true' : 'false', bg: active ? '#e8edf5' : '#2b3038', fg: active ? '#0b0f14' : '#c9d1dc', pick });
    const onBtn = (on, toggle, name) => ({ onStr: on ? 'true' : 'false', onBg: on ? '#5cdb5c' : '#2b3038', onText: on ? '#10240f' : '#a7b1c0', onLabel: on ? 'On' : 'Off', toggle, toggleAria: name + ' on/off' });
    const cards = [];
    cards.push(Object.assign({
      name: 'HPF', color: COL[0], border: sb === 0 ? COL[0] : '#1f2836', select: () => this.setState({ selBand: 0 }),
      types: [12, 24].map(s => chip(s + ' dB/oct', eq.hpf.slope === s, () => this.updCh(id, x => { x.eq.hpf.slope = s; }))),
      knobs: [this.knob(id, 0, 'f', 'Freq', Math.log(eq.hpf.f / 20) / Math.log(50), this.fmtF(eq.hpf.f), COL[0])]
    }, onBtn(eq.hpf.on, () => this.updCh(id, x => { x.eq.hpf.on = !x.eq.hpf.on; }), 'HPF')));
    const typeSets = [[['LS', 'Shelf'], ['PEQ', 'Bell']], [['PEQ', 'Bell']], [['PEQ', 'Bell']], [['PEQ', 'Bell'], ['HS', 'Shelf']]];
    eq.bands.forEach((b, i) => {
      const band = i + 1;
      cards.push(Object.assign({
        name: NAMES[band], color: COL[band], border: sb === band ? COL[band] : '#1f2836', select: () => this.setState({ selBand: band }),
        types: typeSets[i].map(t => chip(t[1], b.type === t[0], () => this.updCh(id, x => { x.eq.bands[i].type = t[0]; }))),
        knobs: [
          this.knob(id, band, 'f', 'Freq', xOf(b.f), this.fmtF(b.f), COL[band]),
          this.knob(id, band, 'g', 'Gain', (b.g + 15) / 30, this.fmtG(b.g), COL[band]),
          this.knob(id, band, 'q', 'Q', Math.log(b.q / 0.3) / Math.log(16 / 0.3), b.q.toFixed(2), COL[band])
        ]
      }, onBtn(b.on, () => this.updCh(id, x => { x.eq.bands[i].on = !x.eq.bands[i].on; }), NAMES[band])));
    });
    let selInfo;
    if (sb === 0) selInfo = 'HPF · ' + this.fmtF(eq.hpf.f) + ' · ' + eq.hpf.slope + ' dB/oct' + (eq.hpf.on ? '' : ' · off');
    else { const b = eq.bands[sb - 1]; selInfo = NAMES[sb] + ' · ' + this.fmtF(b.f) + ' · ' + this.fmtG(b.g) + ' · Q ' + b.q.toFixed(2) + (b.on ? '' : ' · off'); }

      curvePath, fillPath, curveStroke: eq.on ? '#f0d36a' : '#5a6272', curveFill: eq.on ? 'rgba(240,211,106,0.10)' : 'none',
      bandFill, bandFillColor: own.length ? COL[sb] + '2e' : 'none', bandStroke: own.length ? COL[sb] : 'none',
      gridV, gridH, freqLabels, gainLabels, nodes, cards, selInfo,
      eqOnStr: eq.on ? 'true' : 'false', eqOnBg: eq.on ? '#5cdb5c' : '#2b3038', eqOnText: eq.on ? '#10240f' : '#c9d1dc', eqOnLabel: eq.on ? 'EQ In' : 'EQ Bypassed',
      toggleEq: () => this.updCh(id, x => { x.eq.on = !x.eq.on; }),
      flatEq: () => this.updCh(id, x => { x.eq.bands.forEach(b => { b.g = 0; }); })
    };
  }

  dynVals(c, which) {
    const st = this.state, id = c.id, g = which === 'gate', p = g ? c.dyn.gate : c.dyn.comp, lo = g ? -80 : -60, S = 200;
    const map = v => (v - lo) / (-lo) * S;
    const fOut = x => g ? this.gateOut(p, x) : this.compOut(p, x);
    const pts = [];
    for (let i = 0; i <= 80; i++) { const x = lo + (-lo) * i / 80; pts.push(map(x).toFixed(1) + ' ' + (S - this.clamp(map(fOut(x)), -4, S + 4)).toFixed(1)); }
    const lvl = this.clamp(this.levels[id] != null ? this.levels[id] : -90, lo, 0);
    const outL = fOut(lvl);
    const grDb = g ? (p.on && lvl < p.thr ? p.range : 0) : Math.max(0, lvl - (outL - p.makeup));
    const grid = [];
    for (let v = lo + 10; v < 0; v += 10) grid.push({ p: map(v).toFixed(1), q: (S - map(v)).toFixed(1) });
    const color = g ? '#3ed0e0' : '#ff9f43';
    const K = (label, key, prop, def) => this.chKnob(id, label, key, x => x.dyn[which][prop], (x, v) => { x.dyn[which][prop] = v; }, def, color);
    const knobs = g
      ? [K('Threshold', 'gthr', 'thr', -50), K('Range', 'range', 'range', 40), K('Attack', 'gatt', 'att', 0.5), K('Hold', 'hold', 'hold', 50), K('Release', 'grel', 'rel', 150)]
      : [K('Threshold', 'cthr', 'thr', -20), K('Ratio', 'ratio', 'ratio', 3), K('Knee', 'knee', 'knee', 3), K('Attack', 'catt', 'att', 10), K('Release', 'crel', 'rel', 120), K('Makeup', 'makeup', 'makeup', 0)];
    const t = this.toggleBtn(p.on, '#5cdb5c', '#10240f');
    return {
      dvPath: 'M' + pts.join(' L'), dvThrX: map(p.thr).toFixed(1), dvDotX: map(lvl).toFixed(1), dvDotY: (S - this.clamp(map(outL), 0, S)).toFixed(1),
      dvGrid: grid, dvKnobs: knobs, dvColor: color,
      dvFill: 'M' + pts.join(' L') + ' L200 200 L0 200 Z', dvFillColor: color + '1c',
      dvThrL: this.clamp(map(p.thr) / 2, 8, 92).toFixed(1), dvThrText: p.thr.toFixed(1) + ' dB',
      dvDotL: (map(lvl) / 2).toFixed(1), dvDotT: (100 - this.clamp(map(outL), 0, S) / 2).toFixed(1),
      dvAxis: (g ? [-80, -60, -40, -20, 0] : [-60, -45, -30, -15, 0]).map((v, i, arr) => ({ label: String(v), left: (map(v) / 2).toFixed(1), top: (100 - map(v) / 2).toFixed(1), shift: i === 0 ? '0' : i === arr.length - 1 ? '-100%' : '-50%' })),
      dvGrPct: this.clamp(grDb / (g ? 80 : 24) * 100, 0, 100).toFixed(1), dvGrText: grDb > 0.05 ? '-' + grDb.toFixed(1) : '0.0',
      dvOnStr: t.str, dvOnBg: t.bg, dvOnText: t.text, dvOnLabel: (g ? 'Gate ' : 'Compressor ') + (p.on ? 'In' : 'Bypassed'),
      dvToggle: () => { if (!st.locked) this.updCh(id, x => { x.dyn[which].on = !x.dyn[which].on; }); },
      dvSummary: g ? 'Thr ' + p.thr.toFixed(1) + ' dB · Range ' + p.range + ' dB' : 'Thr ' + p.thr.toFixed(1) + ' dB · ' + p.ratio.toFixed(1) + ':1 · Makeup ' + this.fmtG(p.makeup),
      dvAria: (g ? 'Gate' : 'Compressor') + ' transfer curve'
    };
  }

  routingVals() {
    const st = this.state, pad = n => String(n).padStart(2, '0');
    const used = {};
    for (let n = 1; n <= 48; n++) { const c = st.chans['in' + n]; used[c.src] = c.name; }
    const selC = st.chans[st.patchSel];
    const ids = Array.from({ length: 16 }, (_, k) => 'in' + (st.routeBank * 16 + k + 1));
    const chipOf = (on) => ({ pressed: on ? 'true' : 'false', bg: on ? '#e8edf5' : '#2b3038', fg: on ? '#0b0f14' : '#c9d1dc' });
    const next = () => { const n = parseInt(st.patchSel.slice(2), 10); const nn = n % 48 + 1; return { patchSel: 'in' + nn, routeBank: Math.floor((nn - 1) / 16) }; };
    return {
      routeBanks: ['1–16', '17–32', '33–48'].map((l, b) => Object.assign({ label: l, pick: () => this.setState({ routeBank: b }) }, chipOf(st.routeBank === b))),
      routeTiles: ids.map(id => {
        const c = st.chans[id], on = id === st.patchSel;
        return { name: c.name, src: c.src, srcColor: c.src === '—' ? '#ff8a80' : '#5fe3f0', pressed: on ? 'true' : 'false', border: on ? '#ffffff' : '#222a36', bg: on ? '#1b2330' : '#0a0f17', pick: () => this.setState({ patchSel: id }) };
      }),
      patchSelName: selC.name,
      srcGroups: [['Lcl', 'Local'], ['Net', 'Network'], ['USB', 'USB']].map(g => Object.assign({ label: g[1], pick: () => this.setState({ patchGroup: g[0] }) }, chipOf(st.patchGroup === g[0]))),
      srcTiles: Array.from({ length: 32 }, (_, k) => {
        const label = st.patchGroup + ' ' + pad(k + 1), isCur = selC.src === label, u = used[label];
        return {
          label, used: isCur ? 'patched' : (u ? '→ ' + u : ''), pressed: isCur ? 'true' : 'false',
          bg: isCur ? '#5cdb5c' : u ? '#141b26' : '#0a0f17', fg: isCur ? '#10240f' : u ? '#c9d1dc' : '#8792a4', border: isCur ? '#5cdb5c' : '#222a36',
          pick: () => { if (st.locked) return; this.updCh(st.patchSel, x => { x.src = label; }); this.setState(next()); }
        };
      }),
      unpatch: () => { if (!st.locked) this.updCh(st.patchSel, x => { x.src = '—'; }); }
    };
  }

  connectVals(clock) {
    const st = this.state, devs = this.devices();
    const cur = devs.find(d => d.key === st.connected);
    const offline = st.connected === 'offline';
    const ICON = { console: 'M3 5h18v14H3zM7 9v6M11 9v6M15 9v6M19 9v6', io: 'M4 7h16v10H4zM7 12h.01M10 12h.01M13 12h.01M16 12h.01' };
    return {
      isConnected: !!cur,
      connTitle: cur ? 'Connected to ' + cur.name : offline ? 'Working offline' : 'Not connected',
      connSub: cur ? cur.ip + ' · 48 kHz · latency 1.2 ms' : offline ? 'Changes sync when you connect' : 'Pick a console below',
      connHalo: cur ? 'rgba(95,224,96,.18)' : offline ? 'rgba(240,180,74,.18)' : 'rgba(255,74,61,.18)',
      connCardBorder: cur ? '#2c4a2e' : offline ? '#5a4420' : '#5a2420',
      lastScan: st.lastScan,
      scan: () => this.setState({ lastScan: 'Scanned at ' + clock }),
      disconnect: () => this.setState({ connected: null }),
      goOffline: () => this.setState({ connected: 'offline' }),
      devCards: devs.map(d => {
        const on = d.key === st.connected;
        return {
          name: d.name, sub: d.model + ' · ' + d.ip, icon: ICON[d.kind], canConnect: d.kind === 'console',
          status: on ? 'Connected' : d.kind === 'io' ? 'I/O · in use' : 'Available',
          statusColor: on ? '#7be27b' : d.kind === 'io' ? '#8792a4' : '#c9d1dc',
          border: on ? '#2c4a2e' : '#1f2836',
          btnLabel: on ? 'Connected' : 'Connect', btnBg: on ? '#1a2a1b' : '#5cdb5c', btnText: on ? '#7be27b' : '#10240f', btnBorder: on ? '#2c4a2e' : '#5cdb5c',
          act: () => { if (!on) this.setState({ connected: d.key }); }
        };
      })
    };
  }

  settingsVals() {
    const st = this.state;
    const spp = st.spp || parseInt(this.props.stripsPerScreen ?? '12', 10) || 12;
    const seg = (opts, cur, set) => opts.map(o => {
      const on = o[1] === cur;
      return { label: o[0], pressed: on ? 'true' : 'false', bg: on ? '#e8edf5' : '#1b222d', fg: on ? '#0b0f14' : '#c9d1dc', border: on ? '#e8edf5' : '#343d4b', pick: () => set(o[1]) };
    });
    const onoff = (cur, key) => seg([['Off', false], ['On', true]], cur, v => this.setState({ [key]: v }));
    return {
      settingsSections: [
        { title: 'Display', rows: [
          { label: 'Strips per screen', desc: 'Fewer strips are wider and easier to hit. Swipe to reach the rest.', opts: seg([['8', 8], ['12', 12], ['16', 16]], spp, v => this.setState({ spp: v })) },
          { label: 'Meter peak hold', desc: 'Keeps a line at the highest level for 1.5 s.', opts: onoff(st.peakHold, 'peakHold') }
        ] },
        { title: 'Faders', rows: [
          { label: 'Fine control', desc: 'Faders and knobs move a quarter as far per drag.', opts: onoff(st.fine, 'fine') },
          { label: 'Sends on faders', desc: 'Strip faders set sends to the target mix. The master strip shows that mix.', opts: onoff(st.sof, 'sof') },
          { label: 'Send target', desc: 'Which mix the faders control when sends are on.', opts: Array.from({ length: 16 }, (_, k) => k).map(k => seg([[String(k + 1), k]], st.mix, v => this.setState({ mix: v }))[0]) }
        ] },
        { title: 'Safety', rows: [
          { label: 'Mute Enable', desc: 'When off, mute buttons ignore taps.', opts: onoff(st.muteEnable, 'muteEnable') },
          { label: 'Screen lock', desc: 'Blocks faders, knobs, mutes and patching. Unlock from the top bar.', opts: onoff(st.locked, 'locked') },
          { label: 'Scene recall', desc: 'Confirm needs a press on RECALL. Instant loads on tap.', opts: seg([['Confirm', true], ['Instant', false]], st.sceneConfirm, v => this.setState({ sceneConfirm: v })) }
        ] },
        { title: 'About', rows: [
          { label: 'App version', desc: '[VERSION]', opts: [] },
          { label: 'Console', desc: st.connected === 'offline' ? 'Offline' : st.connected ? 'FOH / Monitor · 48 kHz' : 'Not connected', opts: [] }
        ] }
      ]
    };
  }

  renderVals() {
    const st = this.state;
    const banks = this.bankDefs();
    const bank = banks.find(b => b.key === st.bank);
    const mode = st.mode;
    const isMixer = mode === 'mixer', isChannel = mode === 'channel', isScenes = mode === 'scenes';
    const isFx = mode === 'fx', isRouting = mode === 'routing', isConnect = mode === 'connect', isSettings = mode === 'settings';
    const strips = [];
    for (let i = 0; i < 16; i++) { const it = bank.items[i]; strips.push(it ? this.stripVals(it.id) : { real: false, empty: true }); }
    const marks = [[1, '10'], [0.875, '5'], [0.75, '0'], [0.625, '-5'], [0.5, '-10'], [0.375, '-20'], [0.25, '-30'], [0.125, '-50']].map(a => ({ pct: a[0] * 100, label: a[1], tick: a[1] === '0' ? '#e8edf5' : '#566175' }));
    const TINT = { ch1: '#7be27b', ch17: '#7be27b', ch33: '#7be27b', fx: '#7be27b', mix: '#5fe3f0', grp: '#8fb4ff', mtx: '#ea8cf2', dca: '#ff8a80' };
    const bankBtns = banks.map(b => {
      const on = b.key === st.bank;
      return { label: b.label, pressed: on ? 'true' : 'false', bg: on ? b.color : '#161c26', fg: on ? b.text : TINT[b.key], border: on ? b.color : '#2a323f', weight: on ? 600 : 500, pick: () => this.setState({ bank: b.key }) };
    });
    const n = st.spp || parseInt(this.props.stripsPerScreen ?? '12', 10) || 12;
    const gridCols = n >= 16 ? 'repeat(16, minmax(64px, 1fr))' : 'repeat(16, calc((100% - ' + (3 * (n - 1)) + 'px) / ' + n + '))';

    const sel = st.chans[st.sel];
    const TABS = [['input', 'Input'], ['eq', 'HPF + EQ'], ['gate', 'Gate'], ['comp', 'Compressor'], ['sends', 'Sends']];
    const noteOf = k => {
      if (k === 'input') return sel.input ? sel.src + (sel.input.phantom ? ' · 48V' : '') : 'Bus';
      if (k === 'eq') return sel.eq && sel.eq.on ? 'In' : 'Bypassed';
      if (k === 'gate') return sel.dyn && sel.dyn.gate.on ? 'In' : 'Off';
      if (k === 'comp') return sel.dyn && sel.dyn.comp.on ? 'In' : 'Off';
      return sel.sends ? '16 mixes' : 'None';
    };
    const tabs = TABS.map(t => ({ label: t[1], note: noteOf(t[0]), current: st.tab === t[0] ? 'page' : 'false', border: st.tab === t[0] ? sel.color : '#2b3038', bg: st.tab === t[0] ? '#363d48' : '#2b3038', fg: '#eef2f7', pick: () => this.setState({ tab: t[0] }) }));
    const tabLabel = (TABS.find(t => t[0] === st.tab) || TABS[1])[1];

    const order = [];
    banks.forEach(b => b.items.forEach(it => { if (it.eq) order.push(it.id); }));
    const idx = Math.max(0, order.indexOf(st.sel));
    const goTo = d => { const nid = order[(idx + d + order.length) % order.length]; this.setState({ sel: nid }); };

    const blank = { curvePath: '', fillPath: '', curveStroke: 'none', curveFill: 'none', bandFill: '', bandFillColor: 'none', bandStroke: 'none', gridV: [], gridH: [], freqLabels: [], gainLabels: [], nodes: [], cards: [], selInfo: '', eqOnStr: 'false', eqOnBg: '#2b3038', eqOnText: '#c9d1dc', eqOnLabel: 'EQ', toggleEq: () => {}, flatEq: () => {} };
    const dynBlank = { dvFill: '', dvFillColor: 'none', dvThrL: 0, dvThrText: '', dvDotL: 0, dvDotT: 0, dvAxis: [], dvPath: '', dvThrX: 0, dvDotX: 0, dvDotY: 0, dvGrid: [], dvKnobs: [], dvColor: '#ff9f43', dvGrPct: 0, dvGrText: '', dvOnStr: 'false', dvOnBg: '#2b3038', dvOnText: '#c9d1dc', dvOnLabel: '', dvToggle: () => {}, dvSummary: '', dvAria: '' };
    const hasProc = isChannel && !!sel.eq;
    const eqv = hasProc && st.tab === 'eq' ? this.eqVals(sel) : blank;
    const isDynTab = st.tab === 'gate' || st.tab === 'comp';
    const dv = hasProc && isDynTab ? this.dynVals(sel, st.tab) : dynBlank;
    const iv = this.inputVals(sel);
    const sv = this.sendVals(sel);

    const SCENES = this.scenes();
    const pv = st.scenePv, pending = pv !== st.scene;
    const now = new Date();
    const clock = String(now.getHours()).padStart(2, '0') + ':' + String(now.getMinutes()).padStart(2, '0');

    const msId = st.sof ? 'mix' + (st.mix + 1) : 'mainlr';
    const msLvl = this.levels[msId] != null ? this.levels[msId] : -90;
    const ms = Object.assign(this.stripVals(msId), { meterCover2: ((1 - this.dbToPos(msLvl - 1.5)) * 100).toFixed(1), masterLine: st.sof ? '#22d3e6' : '#ffd25a' });

    const tb = on => ({ bg: on ? '#232b37' : '#0d1219', border: on ? '#5a6577' : '#2a323f', cur: on ? 'page' : 'false' });
    const tFx = tb(isFx), tRt = tb(isRouting), tCn = tb(isConnect), tSt = tb(isSettings);
    const offline = st.connected === 'offline';
    const PAGE_TITLES = { scenes: 'Scenes', fx: 'Effects', routing: 'Routing', connect: 'Connection', settings: 'Settings' };
    const backTo = isChannel ? 'mixer' : (st.prevMode === 'channel' ? 'channel' : 'mixer');

    return Object.assign({
      ms, msTag: st.sof ? 'SENDS MASTER' : 'MASTER', msTagColor: st.sof ? '#5fe3f0' : '#ffd25a', msSub: st.sof ? 'Aux bus' : 'Stereo out',
      isMixer, isChannel, isScenes, isFx, isRouting, isConnect, isSettings,
      showBack: !isMixer, backLabel: backTo === 'channel' ? sel.name : 'Mixer',
      goBack: () => this.setState({ mode: backTo }),
      showPageTitle: !!PAGE_TITLES[mode], pageTitle: PAGE_TITLES[mode] || '',
      locked: st.locked, unlock: () => this.setState({ locked: false }),
      goFx: () => this.openPage('fx'), goRouting: () => this.openPage('routing'), goConnect: () => this.openPage('connect'), goSettings: () => this.openPage('settings'),
      fxBg: tFx.bg, fxBorder: tFx.border, fxCur: tFx.cur, rtBg: tRt.bg, rtBorder: tRt.border, rtCur: tRt.cur,
      cnBg: tCn.bg, cnBorder: tCn.border, cnCur: tCn.cur, stBg: tSt.bg, stBorder: tSt.border, stCur: tSt.cur,
      connDot: st.connected && !offline ? '#5fe060' : offline ? '#f0b44a' : '#ff4a3d',
      connAria: st.connected && !offline ? 'Connection: online, select device' : 'Connection: not connected, select device',

      curSceneNum: String(st.scene + 1).padStart(3, '0'), curSceneName: SCENES[st.scene],
      sceneAria: 'Scene ' + (st.scene + 1) + ' ' + SCENES[st.scene] + ', open scene list',
      sceneBtnBg: isScenes ? '#232b37' : '#0d1219', sceneBtnBorder: isScenes ? '#5a6577' : '#2a323f',
      openScenes: () => { if (!isScenes) this.openPage('scenes', { scenePv: st.scene }); },
      sceneRecall: () => { if (pending) this.setState({ scene: pv, mode: backTo }); },
      sceneCards: SCENES.map((nm, i) => ({
        num: String(i + 1).padStart(3, '0'), name: nm, tag: i === st.scene ? 'Current' : '', pressed: i === pv ? 'true' : 'false',
        border: i === pv ? (i === st.scene ? '#7be27b' : '#f0b44a') : '#222a36', bg: i === pv ? '#1b2330' : '#0e1520',
        pick: () => this.setState(st.sceneConfirm ? { scenePv: i } : { scenePv: i, scene: i })
      })),
      pvSceneNum: String(pv + 1).padStart(3, '0'), pvSceneName: SCENES[pv],
      recallTag: pending ? 'Selected · not recalled yet' : 'Current scene', recallTagColor: pending ? '#f0b44a' : '#7be27b',
      recallBarBorder: pending ? '#a87622' : '#222a36', recallBorder: pending ? '#f0b44a' : '#2a323f', recallDisabled: pending ? 'false' : 'true',
      recallBg: pending ? '#f0b44a' : '#161c26', recallText: pending ? '#241703' : '#5d6878',

      gridCols, strips, marks, bankBtns,
      cs: this.stripVals(st.sel),
      prevCh: () => goTo(-1), nextCh: () => goTo(1),
      tabs, tabLabel,
      isEqTab: st.tab === 'eq', isInputTab: st.tab === 'input', isDynTab, isSendsTab: st.tab === 'sends'
    }, eqv, dv, iv, sv,
      isFx ? this.fxVals() : { fxSlots: [], fxTypes: [], fxKnobs: [], fxIsDelay: false },
      isRouting ? this.routingVals() : { routeBanks: [], routeTiles: [], srcGroups: [], srcTiles: [] },
      isConnect ? this.connectVals(clock) : { devCards: [], isConnected: false },
      isSettings ? this.settingsVals() : { settingsSections: [] });
  }
}
